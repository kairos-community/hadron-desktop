# Isotope GRUB theme.
#
# gfxterm theme matching the ly login + waybar palette. Used by the installed
# system's boot menu (grubmenu.cfg switches to gfxterm and `set theme=` to this
# file, which the 09_isotope_grub_theme oem stage places on COS_STATE alongside
# background.tga). desktop-image is resolved relative to this file, so the bare
# filename works once the whole isotope-theme dir lands in themes/isotope/.
#
# The subtitle is a build-time placeholder: this file lives in the COMMON rootfs
# overlay but the subtitle is per-variant, so the Dockerfile substitutes
# @VARIANT_SUBTITLE@ from /etc/isotope/session after kairos-init. Before
# that substitution existed the i3 variant's boot menu claimed to be sway.
#
# Image is TGA, not PNG: GRUB's png.mod rejects valid PNGs with "unsupported
# format", which makes gfxmenu discard the entire theme; tga.mod decodes cleanly.
# Font: "Unifont Regular 16" is the unicode.pf2 that `loadfont unicode` loads on
# the installed system. The live ISO loads our vendored copy from the theme dir.

desktop-image: "background.tga"
title-text: ""
terminal-font: "Unifont Regular 16"

# Brand wordmark (accent lime / muted), under the mark that background.tga
# carries at the top centre (tools/gen-grub-bg.py).
+ label {
    id    = "__brand__"
    top   = 21%
    left  = 0
    width = 100%
    align = "center"
    color = "#8fd12e"
    font  = "Unifont Regular 16"
    text  = "I S O T O P E"
}
+ label {
    top   = 21%+26
    left  = 0
    width = 100%
    align = "center"
    color = "#595b52"
    font  = "Unifont Regular 16"
    text  = "@VARIANT_SUBTITLE@"
}

+ boot_menu {

    item_spacing         = 6
    icon_width           = 0
    icon_height          = 0

}
