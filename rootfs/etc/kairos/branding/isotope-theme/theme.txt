# Isotope GRUB theme, "One light".
#
# gfxterm theme for the installed system's boot menu (grubmenu.cfg switches to
# gfxterm, loads the fonts and `set theme=`s this file, which the
# 09_isotope_grub_theme oem stage places on COS_STATE) and the live ISO
# (auroraboot/live-grub.cfg.tmpl). Paths are relative to this file.
#
# background.tga is the splash's scene asleep: the mark's shell dark, lit only
# on its inner edge by the core. It is drawn by the splash binary
# (tools/gen-grub-bg.py), scaled to the screen's height and anchored left, so
# the mark sits in the left third on any shape of screen. wordmark.tga and the
# nine select_*.tga slices are drawn by the same tool. Everything is in
# percent of the screen, so the wordmark stays over the menu column.
#
# The subtitle is a build-time placeholder: this file lives in the COMMON rootfs
# overlay but the subtitle is per-variant, so the Dockerfile substitutes
# @VARIANT_SUBTITLE@ from /etc/isotope/session after kairos-init.
#
# Images are TGA, not PNG: GRUB's png.mod rejects valid PNGs with "unsupported
# format", which makes gfxmenu discard the entire theme; tga.mod decodes cleanly.
# Fonts: GRUB fonts are 1-bit bitmaps at fixed sizes. inter-medium-18.pf2 and
# -24.pf2 are made from install-gui/fonts/Inter-Medium.ttf with
#   grub-mkfont -s 18 -o inter-medium-18.pf2 Inter-Medium.ttf
# and grub-mkfont names them "Inter Regular 18" and "Inter Regular 24". The
# terminal (e, c) keeps "Unifont Regular 16", the unicode.pf2 GRUB loads anyway.


desktop-image: "background.tga"
desktop-image-scale-method: "fitheight"
desktop-image-h-align: "left"
desktop-color: "#0c0d0b"
title-text: ""
terminal-font: "Unifont Regular 16"

+ image {

}
+ label {
    left  = 58%+2
    top   = 35%+30
    width = 30%
    align = "left"
    color = "#595b52"
    font  = "Inter Regular 18"
    text  = "@VARIANT_SUBTITLE@"
}
